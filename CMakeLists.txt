cmake_minimum_required(VERSION 3.29)
project(RackRandomizer VERSION 1.0.0 LANGUAGES CXX RC)

set(D2RL_SDK_ROOT "" CACHE PATH "Root of D2RLoader PluginSDK 0.4.0")
if(NOT EXISTS "${D2RL_SDK_ROOT}/include/D2RLPlugin/api.h")
  message(FATAL_ERROR "Set D2RL_SDK_ROOT to the D2RLoader PluginSDK 0.4.0 directory.")
endif()
add_library(rack_randomizer SHARED src/plugin.cpp src/plugin.rc)
target_include_directories(rack_randomizer PRIVATE "${D2RL_SDK_ROOT}/include")
target_compile_features(rack_randomizer PRIVATE cxx_std_20)
target_compile_definitions(rack_randomizer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(rack_randomizer PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(rack_randomizer PRIVATE bcrypt)
target_link_options(rack_randomizer PRIVATE /INCREMENTAL:NO /Brepro)
set_target_properties(rack_randomizer PROPERTIES OUTPUT_NAME "d2rl-rack-randomizer" PREFIX "")
